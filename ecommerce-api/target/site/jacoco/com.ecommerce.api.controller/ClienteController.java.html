package com.ecommerce.api.controller;

import com.ecommerce.api.dto.ClienteDTO;
import com.ecommerce.api.security.UsuarioPrincipal;
import com.ecommerce.api.service.ClienteService;
import jakarta.validation.Valid;
import lombok.RequiredArgsConstructor;
import org.springframework.data.domain.Page;
import org.springframework.http.HttpStatus;
import org.springframework.http.ResponseEntity;
import org.springframework.security.access.AccessDeniedException;
import org.springframework.security.access.prepost.PreAuthorize;
import org.springframework.web.bind.annotation.*;
import java.util.List;

@RestController
@RequestMapping("/clientes")
@RequiredArgsConstructor
public class ClienteController {

    private final ClienteService service;

    /**
     * Lista todos os clientes SEM paginação - apenas ADMIN.
     */
    @GetMapping("/todos")
    @PreAuthorize("hasRole('ADMIN')")
    public ResponseEntity<List<ClienteDTO>> listarTodos() {
        return ResponseEntity.ok(service.listarTodos());
    }

    /**
     * Lista clientes COM paginação - apenas ADMIN.
     */
    @GetMapping
    @PreAuthorize("hasRole('ADMIN')")
    public ResponseEntity<Page<ClienteDTO>> listar(
            @RequestParam(defaultValue = "0") int pagina,
            @RequestParam(defaultValue = "10") int tamanho,
            @RequestParam(defaultValue = "id") String ordenarPor,
            @RequestParam(defaultValue = "DESC") String direcao) {
        return ResponseEntity.ok(service.listarComPaginacao(pagina, tamanho, ordenarPor, direcao));
    }

    /**
     * Retorna o próprio perfil de cliente do usuário autenticado.
     */
    @GetMapping("/me")
    public ResponseEntity<ClienteDTO> obterMeuPerfil() {
        UsuarioPrincipal principal = UsuarioPrincipal.getUsuarioAutenticado();
        if (principal == null || principal.getClienteId() == null) {
            throw new AccessDeniedException("Usuário autenticado não possui cliente vinculado");
        }
        return ResponseEntity.ok(service.obterPorId(principal.getClienteId()));
    }

    /**
     * Obtém cliente por ID - ADMIN pode ver qualquer um, USER só o próprio.
     */
    @GetMapping("/{id}")
    public ResponseEntity<ClienteDTO> obter(@PathVariable Long id) {
        exigirAdminOuProprioCliente(id);
        return ResponseEntity.ok(service.obterPorId(id));
    }

    /**
     * Cadastro de cliente feito diretamente pelo admin (sem login associado).
     * O autocadastro normal do cliente acontece em POST /auth/registrar.
     */
    @PostMapping
    @PreAuthorize("hasRole('ADMIN')")
    public ResponseEntity<ClienteDTO> criar(@Valid @RequestBody ClienteDTO dto) {
        return ResponseEntity.status(HttpStatus.CREATED).body(service.criar(dto));
    }

    /**
     * Atualiza cliente - ADMIN pode atualizar qualquer um, USER só o próprio.
     */
    @PutMapping("/{id}")
    public ResponseEntity<ClienteDTO> atualizar(@PathVariable Long id, @Valid @RequestBody ClienteDTO dto) {
        exigirAdminOuProprioCliente(id);
        return ResponseEntity.ok(service.atualizar(id, dto));
    }

    @DeleteMapping("/{id}")
    @PreAuthorize("hasRole('ADMIN')")
    public ResponseEntity<Void> deletar(@PathVariable Long id) {
        service.deletar(id);
        return ResponseEntity.noContent().build();
    }

    private void exigirAdminOuProprioCliente(Long clienteId) {
        UsuarioPrincipal principal = UsuarioPrincipal.getUsuarioAutenticado();
        if (principal == null) {
            throw new AccessDeniedException("Não autenticado");
        }
        if (principal.isAdmin()) {
            return;
        }
        if (!clienteId.equals(principal.getClienteId())) {
            throw new AccessDeniedException("Você só pode acessar seus próprios dados de cliente");
        }
    }
}
