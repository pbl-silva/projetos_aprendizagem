package com.ecommerce.api.security;

import com.ecommerce.api.model.Usuario;
import lombok.AllArgsConstructor;
import lombok.Getter;
import org.springframework.security.core.context.SecurityContextHolder;

/**
 * Representa o usuário autenticado na requisição atual.
 * Fica disponível como "principal" da Authentication após validação do JWT.
 */
@Getter
@AllArgsConstructor
public class UsuarioPrincipal {

    private final Long userId;
    private final String email;
    private final Usuario.Papel papel;
    private final Long clienteId;

    public boolean isAdmin() {
        return papel == Usuario.Papel.ADMIN;
    }

    /**
     * Retorna o usuário autenticado no contexto de segurança atual, ou null
     * caso não haja autenticação (não deveria ocorrer em endpoints protegidos).
     */
    public static UsuarioPrincipal getUsuarioAutenticado() {
        var authentication = SecurityContextHolder.getContext().getAuthentication();
        if (authentication == null || !(authentication.getPrincipal() instanceof UsuarioPrincipal principal)) {
            return null;
        }
        return principal;
    }
}
